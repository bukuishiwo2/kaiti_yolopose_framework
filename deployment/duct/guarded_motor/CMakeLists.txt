cmake_minimum_required(VERSION 3.8)
project(duct_guarded_motor)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(serial_driver REQUIRED)
find_package(tf2 REQUIRED)
find_package(obstacle_dect_msgs REQUIRED)
add_executable(duct_guarded_motor Wheel_Motor_Driver.cpp wheelmotor.cpp)
target_include_directories(duct_guarded_motor PRIVATE include .)
ament_target_dependencies(duct_guarded_motor rclcpp std_msgs sensor_msgs
  geometry_msgs serial_driver tf2 obstacle_dect_msgs)
